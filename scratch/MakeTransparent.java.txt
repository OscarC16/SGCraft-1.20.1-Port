import javax.imageio.ImageIO;
import java.awt.image.BufferedImage;
import java.io.File;

public class MakeTransparent {
    public static void main(String[] args) throws Exception {
        File file = new File("src/main/resources/assets/sgcraft/textures/item/stargate_pad.png");
        BufferedImage img = ImageIO.read(file);
        if (img == null) {
            System.out.println("Could not read image!");
            return;
        }

        int w = img.getWidth();
        int h = img.getHeight();
        BufferedImage dest = new BufferedImage(w, h, BufferedImage.TYPE_INT_ARGB);

        int transparentCount = 0;
        int preservedCount = 0;

        for (int y = 0; y < h; y++) {
            for (int x = 0; x < w; x++) {
                int argb = img.getRGB(x, y);
                int a = (argb >> 24) & 0xFF;
                int r = (argb >> 16) & 0xFF;
                int g = (argb >> 8) & 0xFF;
                int b = argb & 0xFF;

                // If it is light gray or white (r, g, b > 180), make it transparent
                if (r > 180 && g > 180 && b > 180) {
                    dest.setRGB(x, y, 0x00000000); // Fully transparent
                    transparentCount++;
                } else {
                    dest.setRGB(x, y, argb); // Preserve original
                    preservedCount++;
                }
            }
        }

        ImageIO.write(dest, "PNG", file);
        System.out.println("Processed " + file.getName() + ": made " + transparentCount + " pixels transparent, preserved " + preservedCount + " pixels.");
    }
}
